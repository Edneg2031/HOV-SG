# HOV-SG 服务器环境配置与数据检查步骤
#
# 建议逐段复制执行，不要直接整文件一次性运行。项目要求 Python 3.9，
# conda env create 会新建 hovsg 环境，不会修改当前 horizonstream 环境。


# ============================================================
# 1. 处理 conda 的 SSL WRONG_VERSION_NUMBER
# ============================================================

# 先查看服务器是否配置了代理或同时加载多套 .condarc。
env | grep -i proxy
conda config --show-sources
conda config --show | grep -E 'proxy|ssl_verify|channel'

# 当前华为服务器同时加载了：
#   /home/huawei/miniconda3/.condarc -> repo.anaconda.com 外网源
#   /root/.condarc                 -> conda.rnd.huawei.com 内网源
# 外网源经过公司代理时 TLS 握手失败。用一个临时 CONDARC 隔离两套配置，
# 仅通过华为内网 Conda 代理解析 defaults、pytorch 等频道。
HOVSG_CONDARC=$(mktemp)
conda config --file "$HOVSG_CONDARC" --set ssl_verify true
conda config --file "$HOVSG_CONDARC" --set channel_alias http://conda.rnd.huawei.com/repository/conda-proxy
conda config --file "$HOVSG_CONDARC" --add default_channels http://conda.rnd.huawei.com/repository/conda-proxy/main
conda config --file "$HOVSG_CONDARC" --add default_channels http://conda.rnd.huawei.com/repository/conda-proxy/r
conda config --file "$HOVSG_CONDARC" --add channels defaults
conda config --file "$HOVSG_CONDARC" --add channels pytorch
conda config --file "$HOVSG_CONDARC" --set channel_priority strict
export CONDARC="$HOVSG_CONDARC"

# 确认输出只包含 conda.rnd.huawei.com，不应再出现 repo.anaconda.com：
conda config --show-sources
conda config --show channels default_channels channel_alias

conda clean --index-cache -y
conda env create -f environment.yaml

# 如果不再需要临时配置，可在全部安装完成后执行：
# unset CONDARC
# rm -f "$HOVSG_CONDARC"

# 注意：不要把带有用户名和密码的代理 URL 粘贴到日志、Git 或聊天中。
# 如果已经公开，立即删除消息并更换密码。关闭 ssl_verify 对当前
# WRONG_VERSION_NUMBER 没有帮助，不建议使用。

# ============================================================
# 5. 检查 HM3D-Omni 是否能直接作为 HOV-SG 输入
# ============================================================

SCENE_DIR=data/hm3d_omni/00127-EN7GiDgxdQ2

# 查看各目录的文件后缀、数量和几个示例路径：
for MODALITY in rgb depth_zbuffer depth_euclidean mask_valid point_info; do
  echo "===== $MODALITY ====="
  find "$SCENE_DIR/$MODALITY" -type f | sed 's/.*\.//' | sort | uniq -c
  find "$SCENE_DIR/$MODALITY" -type f | sort | head -5
done

# 查看 point_info 中普通文本或 JSON 文件的内容（如果存在）：
find "$SCENE_DIR/point_info" -type f \( -name '*.txt' -o -name '*.json' \) \
  -print -exec head -30 {} \; | head -100

# HOV-SG 当前的 hm3dsem loader 要求每一帧严格对应：
#   rgb/<frame>.png
#   depth/<frame>.png
#   pose/<frame>.txt       （16 个浮点数，表示 4x4 相机位姿）
#
# 当前场景解压结果为：
#   rgb              1150 files
#   depth_zbuffer    21613 files
#   depth_euclidean   4041 files
#   mask_valid       21613 files
#   point_info       21613 files
#
# 数量不一致，且 point_info 不等于已经确认可用的 4x4 pose，因此不能直接：
#   mv point_info pose
# 也不能直接运行 application/create_graph.py。
#
# 下一步必须根据上面文件示例确认：
#   1. rgb、depth 和 point_info 的文件命名/目录层级如何关联；
#   2. point_info 是否包含 camera_to_world/world_to_camera、位置和四元数；
#   3. 深度的文件格式、单位和相机内参；
#   4. 从 21613 个深度文件中选择与 1150 张 RGB 对应的帧。
# 确认这些字段后，再编写转换脚本生成 HOV-SG 的 rgb/depth/pose 目录。
