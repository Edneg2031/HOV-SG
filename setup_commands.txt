# 1. 处理 conda 的 SSL WRONG_VERSION_NUMBER

#   /home/huawei/miniconda3/.condarc -> repo.anaconda.com 外网源
#   /root/.condarc                 -> conda.rnd.huawei.com 内网源
# 安装目录级 .condarc 仍会被合并，因此不使用 conda env create。通过
# --override-channels 明确限制为华为内网源，先安装 YAML 中的 Conda 依赖。

conda create -n hovsg -y --override-channels \
  -c http://conda.rnd.huawei.com/repository/conda-proxy/pytorch \
  -c http://conda.rnd.huawei.com/repository/conda-proxy/main \
  python=3.9 numpy faiss-gpu pip

# WRONG_VERSION_NUMBER 没有帮助，不建议使用。

# ============================================================
# 5. 检查 HM3D-Omni 是否能直接作为 HOV-SG 输入
# ============================================================

SCENE_DIR=data/hm3d_omni/00127-EN7GiDgxdQ2

# 查看各目录的文件后缀、数量和几个示例路径：
for MODALITY in rgb depth_zbuffer depth_euclidean mask_valid point_info; do
  echo "===== $MODALITY ====="
  find "$SCENE_DIR/$MODALITY" -type f | sed 's/.*\.//' | sort | uniq -c
  find "$SCENE_DIR/$MODALITY" -type f | sort | head -5
done

# 查看 point_info 中普通文本或 JSON 文件的内容（如果存在）：
find "$SCENE_DIR/point_info" -type f \( -name '*.txt' -o -name '*.json' \) \
  -print -exec head -30 {} \; | head -100

# HOV-SG 当前的 hm3dsem loader 要求每一帧严格对应：
#   rgb/<frame>.png
#   depth/<frame>.png
#   pose/<frame>.txt       （16 个浮点数，表示 4x4 相机位姿）
#
# 当前场景解压结果为：
#   rgb              1150 files
#   depth_zbuffer    21613 files
#   depth_euclidean   4041 files
#   mask_valid       21613 files
#   point_info       21613 files
#
# 数量不一致，且 point_info 不等于已经确认可用的 4x4 pose，因此不能直接：
#   mv point_info pose
# 也不能直接运行 application/create_graph.py。
#
# 下一步必须根据上面文件示例确认：
#   1. rgb、depth 和 point_info 的文件命名/目录层级如何关联；
#   2. point_info 是否包含 camera_to_world/world_to_camera、位置和四元数；
#   3. 深度的文件格式、单位和相机内参；
#   4. 从 21613 个深度文件中选择与 1150 张 RGB 对应的帧。
# 确认这些字段后，再编写转换脚本生成 HOV-SG 的 rgb/depth/pose 目录。
